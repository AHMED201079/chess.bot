class GroupDict:
    """Dictionary facade: every group gets its own independent namespace."""
    def __init__(self, filename, default_factory=dict):
        self.filename = filename
        self.default_factory = default_factory
        self.data = self._load()

    def _load(self):
        if not os.path.exists(self.filename):
            return {_GROUP_KEY: {}}
        try:
            with open(self.filename, "r", encoding="utf-8") as f:
                raw = json.load(f)
        except Exception:
            return {_GROUP_KEY: {}}

        if not isinstance(raw, dict) or _GROUP_KEY not in raw or not isinstance(raw.get(_GROUP_KEY), dict):
            return {_GROUP_KEY: {}, "__legacy__": raw}
        return raw

    def _group(self):
        gid = get_active_chat_id()
        groups = self.data.setdefault(_GROUP_KEY, {})
        if gid not in groups:
            groups[gid] = self.default_factory()
        return groups[gid]

    def save(self):
        with open(self.filename, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=4)

    def get(self, key, default=None):
        return self._group().get(key, default)

    def items(self):
        return self._group().items()

    def values(self):
        return self._group().values()

    def keys(self):
        return self._group().keys()

    def __contains__(self, key):
        return key in self._group()

    def __getitem__(self, key):
        return self._group()[key]

    def __setitem__(self, key, value):
        self._group()[key] = value

    def __delitem__(self, key):
        del self._group()[key]

    def __bool__(self):
        return bool(self._group())

    def __len__(self):
        return len(self._group())

class GroupList(GroupDict):
    def __init__(self, filename):
        super().__init__(filename, list)

    def _group(self):
        gid = get_active_chat_id()
        groups = self.data.setdefault(_GROUP_KEY, {})
        if gid not in groups:
            groups[gid] = []
        return groups[gid]

    def append(self, value):
        if value not in self._group():
            self._group().append(value)

    def remove(self, value):
        if value in self._group():
            self._group().remove(value)

    def __iter__(self):
        return iter(self._group())

def migrate_legacy_here():
    gid = get_active_chat_id()
    changed = False
    for store in (users, points, matches, daily_wins, badges):
        legacy = store.data.get("__legacy__")
        if legacy and isinstance(legacy, dict):
            store.data.setdefault(_GROUP_KEY, {})[gid] = legacy
            del store.data["__legacy__"]
            store.save()
            changed = True
    legacy = point_admins.data.get("__legacy__")
    if isinstance(legacy, list):
        point_admins.data.setdefault(_GROUP_KEY, {})[gid] = legacy
        del point_admins.data["__legacy__"]
        point_admins.save()
        changed = True
    return changed

def get_lichess_puzzle(difficulty="medium"):
    rating_ranges = {
        "easy": (600, 1100),
        "medium": (1100, 1600),
        "hard": (1600, 2100),
        "extreme": (2100, 3000)
    }
    low, high = rating_ranges.get(difficulty, (1100, 1600))
    
    try:
        for _ in range(5):
            response = requests.get("https://lichess.org/api/puzzle/next", timeout=10)
            if response.status_code == 200:
                data = response.json()
                puzzle_rating = data.get("puzzle", {}).get("rating", 1500)
                if low <= puzzle_rating <= high:
                    return data
        return data if response.status_code == 200 else None
    except Exception as e:
        print("Lichess puzzle error:", e)
        return None

def get_puzzle_board(data):
    try:
        game_pgn = data["game"]["pgn"]
        initial_ply = data["puzzle"]["initialPly"]
        pgn = io.StringIO(game_pgn)
        game = chess.pgn.read_game(pgn)
        if game is None:
            return None
        board = game.board()
        for ply, move in enumerate(game.mainline_moves()):
            if ply >= initial_ply:
                break
            board.push(move)
        return board
    except Exception as e:
        print("Puzzle board error:", e)
        return None

def render_puzzle_board(board):
    try:
        board_size = 720
        square_size = board_size // 8
        light_square = (240, 217, 181)
        dark_square = (181, 136, 99)

        image = Image.new("RGB", (board_size, board_size), light_square)
        draw = ImageDraw.Draw(image)

        for rank in range(8):
            for file in range(8):
                x1 = file * square_size
                y1 = rank * square_size
                x2 = x1 + square_size
                y2 = y1 + square_size
                if (rank + file) % 2 == 1:
                    draw.rectangle([x1, y1, x2, y2], fill=dark_square)

        pieces = {
            "P": "♙", "N": "♘", "B": "♗", "R": "♖", "Q": "♕", "K": "♔",
            "p": "♟", "n": "♞", "b": "♝", "r": "♜", "q": "♛", "k": "♚",
        }

        try:
            piece_font = ImageFont.truetype("/system/fonts/NotoSansSymbols2-Regular.ttf", 78)
        except Exception:
            piece_font = ImageFont.load_default()

        for square, piece in board.piece_map().items():
            file = chess.square_file(square)
            rank = 7 - chess.square_rank(square)
            x = file * square_size
            y = rank * square_size
            symbol = pieces[piece.symbol()]

            bbox = draw.textbbox((0, 0), symbol, font=piece_font)
            text_width = bbox[2] - bbox[0]
            text_height = bbox[3] - bbox[1]
            text_x = x + (square_size - text_width) / 2
            text_y = y + (square_size - text_height) / 2 - 8

            draw.text((text_x, text_y), symbol, font=piece_font, fill=(20, 20, 20))

        output = io.BytesIO()
        output.name = "puzzle.png"
        image.save(output, format="PNG")
        output.seek(0)
        return output
    except Exception as e:
        print("Puzzle render error:", e)
        return None

def get_elo(username):
    url = f"https://api.chess.com/pub/player/{username}/stats"
    headers = {"User-Agent": "TelegramChessBot/1.0 (contact: admin@example.com)"}
    try:
        r = requests.get(url, headers=headers, timeout=10)
        if r.status_code != 200:
            return {"rapid": "N/A", "blitz": "N/A", "bullet": "N/A"}
        data = r.json()
    except:
        return {"rapid": "Error", "blitz": "Error", "bullet": "Error"}

    def safe(key):
        try: return data.get(key, {}).get("last", {}).get("rating", "N/A")
        except: return "N/A"
    return {"rapid": safe("chess_rapid"), "blitz": safe("chess_blitz"), "bullet": safe("chess_bullet")}

def get_games(username):
    url = f"https://api.chess.com/pub/player/{username}/games/archives"
    try:
        r = requests.get(url, timeout=10)
        if r.status_code != 200: return []
        archives = r.json().get("archives", [])
        all_games = []
        for archive in archives[-2:]:
            try:
                game_request = requests.get(archive, timeout=10)
                if game_request.status_code == 200:
                    all_games.extend(game_request.json().get("games", []))
            except: continue
        return all_games
    except: return []

def get_match_result(player1, player2, challenge_time):
    games = get_games(player1)
    for game in reversed(games):
        end_time = game.get("end_time", 0)
        if end_time < challenge_time: continue
        white = game["white"]["username"].lower()
        black = game["black"]["username"].lower()
        if {white, black} != {player1.lower(), player2.lower()}: continue

        if game["white"]["result"] == "win":
            return {"winner": white, "game_url": game.get("url", "")}
        if game["black"]["result"] == "win":
            return {"winner": black, "game_url": game.get("url", "")}
        return {"winner": "draw", "game_url": game.get("url", "")}
    return None

async def handle_puzzle_reply(update: Update, context: ContextTypes.DEFAULT_TYPE):
    set_active_chat(update.effective_chat.id)
    chat_id = str(update.effective_chat.id)
    msg = update.message
    
    if not msg or not msg.text or not msg.text.startswith("/"):
        return
        
    if chat_id not in puzzles or puzzles[chat_id].get("status") != "active":
        return

    current_puzzle = puzzles[chat_id]
    
    elapsed_minutes = (time.time() - current_puzzle["started_at"]) / 60
    if elapsed_minutes > current_puzzle["time_limit"]:
        finish_puzzle(chat_id)
        await msg.reply_text("⏱️ <b>انتهى الوقت المحدد للغز!</b> لم تعد الإجابات مقبولة.", parse_mode="HTML")
        return

    if msg.from_user.id != current_puzzle["owner_id"]:
        return

    user_move_str = msg.text.replace("/", "").strip()
    board = chess.Board(current_puzzle["fen"])
    
    try:
        user_move = board.parse_san(user_move_str)
    except:
        return

    expected_move_str = current_puzzle["solution"][current_puzzle["current_step"]]
    expected_move = board.parse_san(expected_move_str)

    if user_move == expected_move:
        board.push(user_move)
        current_puzzle["current_step"] += 1
        
        if current_puzzle["current_step"] >= len(current_puzzle["solution"]):
            finish_puzzle(chat_id)
            board_img = render_puzzle_board(board)
            await msg.reply_photo(
                photo=board_img,
                caption="🎉 <b>أحسنت! إجابة صحيحة تماماً وتم حل اللغز بنجاح!</b> 🏆",
                parse_mode="HTML"
            )
        else:
            reply_move_str = current_puzzle["solution"][current_puzzle["current_step"]]
            reply_move = board.parse_san(reply_move_str)
            board.push(reply_move)
            current_puzzle["current_step"] += 1
            
            current_puzzle["fen"] = board.fen()
            puzzles[chat_id] = current_puzzle
            save_puzzles()
            
            if current_puzzle["current_step"] >= len(current_puzzle["solution"]):
                finish_puzzle(chat_id)
                board_img = render_puzzle_board(board)
                await msg.reply_photo(
                    photo=board_img,
                    caption="🎉 <b>أحسنت! تم إكمال اللغز بنجاح!</b> 🏆",
                    parse_mode="HTML"
                )
            else:
                board_img = render_puzzle_board(board)
                await msg.reply_photo(
                    photo=board_img,
                    caption=f"✅ <b>نقلة صحيحة!</b> وتم رد الكمبيوتر بـ <code>{reply_move_str}</code>.\nما هي نقلتك التالية؟",
                    parse_mode="HTML"
                )
    else:
        current_puzzle["wrong_attempts"] += 1
        if current_puzzle["wrong_attempts"] >= 2:
            sol_text = " ➡️ ".join(current_puzzle["solution"])
            finish_puzzle(chat_id)

async def result(update: Update, context: ContextTypes.DEFAULT_TYPE):
    set_active_chat(update.effective_chat.id)
    if not update.message.reply_to_message:
        await update.message.reply_text("❌ استخدم الأمر بالرد على رسالة التحدي.")
        return
    if not await is_admin(update, update.effective_user.id) and str(update.effective_user.id) not in point_admins:
        await update.message.reply_text("❌ هذا الأمر للمشرفين ومسؤولي النقاط فقط.")
        return

    reply_id = update.message.reply_to_message.message_id
    match = None
    for data in matches.values():
        if data.get("message_id") == reply_id:
            match = data
            break
    if not match:
        await update.message.reply_text("❌ هذه ليست رسالة تحدي.")
        return

    res = get_match_result(match["username1"], match["username2"], match["time"])
    if res is None:
        await update.message.reply_text("⏳ لم يتم العثور على مباراة بين اللاعبين بعد.")
        return
    if match["status"] == "finished":
        await update.message.reply_text("❌ تم اعتماد نتيجة هذه المباراة مسبقًا.")
        return

    if res["winner"] == "draw":
        match["status"], match["result"] = "finished", "draw"
        save_matches()
        await context.bot.edit_message_text(chat_id=update.effective_chat.id, message_id=match["message_id"], parse_mode="HTML",
                                            text=f"🤝 <b>انتهت المباراة</b>\n━━━━━━━━━━━━━━\n\n⚖️ <b>النتيجة:</b> تعادل\n\n🔗 <a href=\"{res['game_url']}\">عرض المباراة</a>\n\n✅ تم اعتماد النتيجة رسميًا.")
        return

    if res["winner"] == match["username1"].lower():
        winner_id, loser_id = match["player1"], match["player2"]
        winner_name, loser_name = match["username1"], match["username2"]
    else:
        winner_id, loser_id = match["player2"], match["player1"]
        winner_name, loser_name = match["username2"], match["username1"]

    winner_old = points.get(winner_id, 20)
    points[winner_id] = winner_old + 8
    loser_old = points.get(loser_id, 20)
    lose_points = 8 if loser_old >= 200 else (5 if loser_old >= 100 else 0)
    points[loser_id] = max(0, loser_old - lose_points)
    save_points()

    today = datetime.now().strftime("%Y-%m-%d")
    if today not in daily_wins: daily_wins[today] = {}
    daily_wins[today][winner_id] = daily_wins[today].get(winner_id, 0) + 1
    save_daily_wins()

    match["status"], match["result"] = "finished", res["winner"]
    save_matches()

    await context.bot.edit_message_text(chat_id=update.effective_chat.id, message_id=match["message_id"], parse_mode="HTML",
                                        text=f"🏁 <b>انتهت المباراة</b>\n━━━━━━━━━━━━━━\n\n👑 <b>الفائز</b>\n<code>{winner_name}</code>\n🏆 <b>المكافأة:</b> +8 نقطة\n\n📉 <b>الخاسر</b>\n<code>{loser_name}</code>\n➖ <b>الخصم:</b> {lose_points} نقطة\n\n🔗 <a href=\"{res['game_url']}\">عرض المباراة</a>\n\n✅ تم اعتماد النتيجة رسميًا.")
    await update.message.reply_text("✅ تم اعتماد النتيجة وتحديث النقاط بنجاح.")
