def get_elo(username):
    url = f"https://api.chess.com/pub/player/{username}/stats"
    headers = {
        "User-Agent": "TelegramChessBot/1.0 (contact: admin@example.com)"
    }

    try:
        r = requests.get(url, headers=headers, timeout=10)
        if r.status_code != 200:
            return {"rapid": "N/A", "blitz": "N/A", "bullet": "N/A"}
        data = r.json()
    except:
        return {"rapid": "Error", "blitz": "Error", "bullet": "Error"}

    def safe(key):
        try:
            return data.get(key, {}).get("last", {}).get("rating", "N/A")
        except:
            return "N/A"

    return {
        "rapid": safe("chess_rapid"),
        "blitz": safe("chess_blitz"),
        "bullet": safe("chess_bullet")
    }

def get_rank(score):
    if score < 50:
        return "♙ مبتدئ"
    elif score < 150:
        return "♘ لاعب نشط"
    elif score < 300:
        return "♗ لاعب متقدم"
    elif score < 500:
        return "♖ خبير"
    else:
        return "👑 أسطورة"

def get_name(user):

async def is_admin(update: Update, user_id: int):
    try:
        member = await update.effective_chat.get_member(user_id)
        return member.status in ["administrator", "creator"]
    except:
        return False

async def handle_points(update: Update, context: ContextTypes.DEFAULT_TYPE):
    msg = update.message

    if not msg or not msg.text:
        return

    text = msg.text.strip()

    if text not in ["+10", "-10"] and not text.startswith("+") and not text.startswith("-"):
        return

    try:
        value = int(text)
    except:
        return

    if not msg.reply_to_message:
        return

    admin_id = update.effective_user.id

    if not await is_admin(update, admin_id) and str(admin_id) not in point_admins:
        return

    user = msg.reply_to_message.from_user
    uid = str(user.id)

    points[uid] = points.get(uid, 20) + value
    save_points()

    await msg.reply_text(
    f"""🏆 تم تعديل النقاط

👤 الاسم: {get_name(user)}
📊 النقاط: {points[uid]}
🎖️ الرتبة: {get_rank(points[uid])}"""
)

async def play(update: Update, context: ContextTypes.DEFAULT_TYPE):
    # التأكد من أن الأمر أُرسل بالرد على رسالة
    if not update.message.reply_to_message:
        await update.message.reply_text("❌ استخدم الأمر بالرد على رسالة اللاعب الذي تريد تحديه.")
        return

    challenger = update.effective_user
    opponent = update.message.reply_to_message.from_user

    challenger_id = str(challenger.id)
    opponent_id = str(opponent.id)

    # منع التحدي مع البوتات أو منع تحدي الشخص لنفسه
    if challenger.is_bot or opponent.is_bot:
        return

    if challenger_id == opponent_id:
        await update.message.reply_text("❌ لا يمكنك تحدي نفسك!")
        return

    # التحقق من أن المتحدي رابط حسابه
    if challenger_id not in users:
        await update.message.reply_text("❌ يجب ربط حسابك أولاً باستخدام /sign")
        return

    # التحقق من أن الطرف الآخر رابط حسابه
    if opponent_id not in users:
        await update.message.reply_text("❌ هذا اللاعب غير مربوط بحساب Chess.com.")
        return

    import time

    match_id = str(int(time.time()))

    matches[match_id] = {
        "player1": challenger_id,
        "player2": opponent_id,
        "username1": users[challenger_id],
        "username2": users[opponent_id],
        "time": int(time.time()),
        "status": "pending"
    }

    save_matches()

    challenger_chess = users[challenger_id]
    opponent_chess = users[opponent_id]

    challenge_url = f"https://www.chess.com/play/online/new?opponent={opponent_chess}"

    text = (
        f"⚔️ <b>تحدي شطرنج جديد!</b>\n\n"
        f"👤 <b>المتحدي:</b> {escape(get_name(challenger))} (<code>{challenger_chess}</code>)\n"
        f"🆚\n"
        f"👤 <b>الخصم:</b> {escape(get_name(opponent))} (<code>{opponent_chess}</code>)\n\n"
        f"🔗 <a href='{challenge_url}'><b>اضغط هنا لبدء التحدي واختيار الوقت</b></a>\n\n"
        f"♟️ بعد انتهاء المباراة، استخدم أمر <code>/result</code> لاعتماد النتيجة."
    )

    await update.message.reply_text(
        text,
        parse_mode="HTML",
        disable_web_page_preview=False
    )
