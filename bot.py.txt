class GroupDict:
    """واجهة قاموس: كل مجموعة لها مساحة بيانات مستقلة تماماً."""

    def __init__(self, filename, default_factory=dict):
        self.filename = filename
        self.default_factory = default_factory
        self.data = self._load()

    def _load(self):
        if not os.path.exists(self.filename):
            return {_GROUP_KEY: {}}
        try:
            with open(self.filename, "r", encoding="utf-8") as f:
                raw = json.load(f)
        except (OSError, json.JSONDecodeError) as exc:
            log.warning("تعذّر قراءة %s (%s) — سيتم البدء ببيانات فارغة.", self.filename, exc)
            return {_GROUP_KEY: {}}

        if not isinstance(raw, dict):
            return {_GROUP_KEY: {}, _LEGACY_KEY: raw}
        if not isinstance(raw.get(_GROUP_KEY), dict):
            return {_GROUP_KEY: {}, _LEGACY_KEY: raw}
        return raw

    def _group(self):
        gid = get_active_chat_id()
        groups = self.data.setdefault(_GROUP_KEY, {})
        current = groups.get(gid)
        if not isinstance(current, type(self.default_factory())):
            current = self.default_factory()
            groups[gid] = current
        return current

    def save(self):
        """كتابة ذرية: نكتب في ملف مؤقت ثم نستبدل، حتى لا تتلف البيانات."""
        with _IO_LOCK:
            directory = os.path.dirname(self.filename) or "."
            fd, tmp = tempfile.mkstemp(dir=directory, suffix=".tmp")
            try:
                with os.fdopen(fd, "w", encoding="utf-8") as f:
                    json.dump(self.data, f, ensure_ascii=False, indent=2)
                os.replace(tmp, self.filename)
            except Exception:
                if os.path.exists(tmp):
                    os.remove(tmp)
                raise

    # واجهة القاموس
    def get(self, key, default=None):
        return self._group().get(key, default)

    def setdefault(self, key, default=None):
        return self._group().setdefault(key, default)

    def pop(self, key, default=None):
        return self._group().pop(key, default)

    def items(self):
        return list(self._group().items())

    def values(self):
        return list(self._group().values())

    def keys(self):
        return list(self._group().keys())

    def __contains__(self, key):
        return key in self._group()

    def __getitem__(self, key):
        return self._group()[key]

    def __setitem__(self, key, value):
        self._group()[key] = value

    def __delitem__(self, key):
        del self._group()[key]

    def __bool__(self):
        return bool(self._group())

    def __len__(self):
        return len(self._group())

    def __iter__(self):
        return iter(self._group())


class GroupList(GroupDict):
    def __init__(self, filename):
        super().__init__(filename, list)

    def append(self, value):
        group = self._group()
        if value not in group:
            group.append(value)

    def remove(self, value):
        group = self._group()
        if value in group:
            group.remove(value)

RATING_RANGES = {
    "easy": (400, 1100),
    "medium": (1100, 1600),
    "hard": (1600, 2100),
    "extreme": (2100, 3200),
}


def fetch_lichess_puzzle(difficulty="medium"):
    """يُنفَّذ في خيط منفصل — لا يحجب حلقة الأحداث."""
    low, high = RATING_RANGES.get(difficulty, RATING_RANGES["medium"])
    headers = {"User-Agent": USER_AGENT, "Accept": "application/json"}
    fallback = None
    for _ in range(6):
        try:
            response = requests.get(
                "https://lichess.org/api/puzzle/next",
                headers=headers,
                timeout=HTTP_TIMEOUT,
            )
            if response.status_code != 200:
                continue
            data = response.json()
        except (requests.RequestException, ValueError) as exc:
            log.warning("Lichess puzzle error: %s", exc)
            continue

        rating = data.get("puzzle", {}).get("rating", 1500)
        if fallback is None:
            fallback = data
        if low <= rating <= high:
            return data
    return fallback


def get_puzzle_board(data):
    """
    مهم: موضع اللغز في Lichess هو الموضع بعد لعب (initialPly + 1) نصف نقلة،
    وليس initialPly كما كان في الكود القديم. تم التحقق عملياً من الـ API.
    """
    try:
        game_pgn = data["game"]["pgn"]
        initial_ply = int(data["puzzle"]["initialPly"])
        game = chess.pgn.read_game(io.StringIO(game_pgn))
        if game is None:
            return None
        board = game.board()
        for index, move in enumerate(game.mainline_moves()):
            board.push(move)
            if index >= initial_ply:
                break
        # تحقق: أول نقلة في الحل يجب أن تكون قانونية في هذا الموضع
        solution = data["puzzle"].get("solution") or []
        if solution:
            try:
                if chess.Move.from_uci(solution[0]) not in board.legal_moves:
                    return None
            except ValueError:
                return None
        return board
    except (KeyError, TypeError, ValueError) as exc:
        log.warning("Puzzle board error: %s", exc)
        return None

def _piece_path(symbol):
    prefix = "w" if symbol.isupper() else "b"
    return os.path.join(PIECE_DIR, f"{prefix}{symbol.upper()}.png")


def download_piece_images(force=False):
    """ينزّل صور القطع مرة واحدة ويخزنها محلياً. يُنفَّذ في خيط منفصل."""
    os.makedirs(PIECE_DIR, exist_ok=True)
    all_ok = True
    for symbol, name in PIECE_FILES.items():
        path = _piece_path(symbol)
        if not force and os.path.exists(path) and os.path.getsize(path) > 500:
            continue
        try:
            response = requests.get(
                PIECE_SOURCE.format(name=name),
                headers={"User-Agent": USER_AGENT},
                timeout=HTTP_TIMEOUT,
            )
            if response.status_code == 200 and response.content[:4] == b"\x89PNG":
                with open(path, "wb") as handle:
                    handle.write(response.content)
            else:
                all_ok = False
        except (requests.RequestException, OSError) as exc:
            log.warning("تعذّر تنزيل صورة القطعة %s: %s", symbol, exc)
            all_ok = False
    if all_ok:
        log.info("صور القطع جاهزة في %s", PIECE_DIR)
    return all_ok


def _piece_image(symbol, size):
    key = (symbol, size)
    if key in _PIECE_IMG_CACHE:
        return _PIECE_IMG_CACHE[key]
    path = _piece_path(symbol)
    if not os.path.exists(path):
        return None
    try:
        image = Image.open(path).convert("RGBA")
        image = image.resize((size, size), Image.LANCZOS)
    except (OSError, ValueError):
        return None
    _PIECE_IMG_CACHE[key] = image
    return image


_SPRITE_CACHE = {}


def _piece_sprite(symbol, size):
    """قطعة جاهزة بظل مدمج مسبقاً (يُحسب مرة واحدة فقط)."""
    key = (symbol, size)
    if key in _SPRITE_CACHE:
        return _SPRITE_CACHE[key]
    base = _piece_image(symbol, size)
    if base is None:
        return None
    pad = 6
    sprite = Image.new("RGBA", (size + pad * 2, size + pad * 2), (0, 0, 0, 0))
    shadow = Image.new("RGBA", sprite.size, (0, 0, 0, 0))
    shadow.paste((0, 0, 0, 110), (pad, pad + 4), base.split()[3])
    shadow = shadow.filter(ImageFilter.GaussianBlur(3.5))
    sprite.alpha_composite(shadow)
    sprite.alpha_composite(base, (pad, pad))
    _SPRITE_CACHE[key] = sprite
    return sprite


def _load_piece_font(size):
    """خط احتياطي لرموز الشطرنج لو تعذّر تنزيل صور القطع."""
    if size in _PIECE_FONT_CACHE:
        return _PIECE_FONT_CACHE[size]
    result = (None, False)
    for path in FONT_CANDIDATES:
        if not os.path.exists(path):
            continue
        try:
            font = ImageFont.truetype(path, size)
            bbox = font.getbbox("\u265b")
            if bbox and (bbox[2] - bbox[0]) > size * 0.25:
                result = (font, True)
                break
            if result[0] is None:
                result = (font, False)
        except OSError:
            continue
    if result[0] is None:
        result = (ImageFont.load_default(), False)
    _PIECE_FONT_CACHE[size] = result
    return result


def _wood_tile(size, base, seed):
    """يولّد مربعاً بملمس خشبي واقعي (عروق + تدرج)."""
    rnd = random.Random(seed)
    tile = Image.new("RGB", (size, size), base)
    draw = ImageDraw.Draw(tile)

    for _ in range(int(size * 1.6)):
        y0 = rnd.uniform(-size * 0.1, size * 1.1)
        shade = rnd.randint(-18, 14)
        color = tuple(max(0, min(255, channel + shade)) for channel in base)
        amplitude = rnd.uniform(1.0, 4.0)
        frequency = rnd.uniform(0.8, 2.4)
        phase = rnd.uniform(0, 6.28)
        points = []
        steps = 10
        for index in range(steps + 1):
            x = size * index / steps
            y = y0 + math.sin(phase + frequency * math.pi * index / steps) * amplitude
            points.append((x, y))
        draw.line(points, fill=color, width=rnd.choice([1, 1, 2]))

    # عقد خشبية خفيفة
    for _ in range(rnd.randint(0, 2)):
        cx, cy = rnd.uniform(0, size), rnd.uniform(0, size)
        for ring in range(rnd.randint(3, 6)):
            radius = 3 + ring * rnd.uniform(2.5, 4.5)
            shade = rnd.randint(-22, -6)
            color = tuple(max(0, min(255, channel + shade)) for channel in base)
            draw.ellipse(
                [cx - radius, cy - radius * 0.7, cx + radius, cy + radius * 0.7],
                outline=color, width=1,
            )

    tile = tile.filter(ImageFilter.GaussianBlur(0.7))

    # إضاءة ناعمة من أعلى اليسار
    shading = Image.new("L", (size, size))
    shade_draw = ImageDraw.Draw(shading)
    for i in range(size):
        shade_draw.line([(0, i), (size, i)], fill=int(128 + 26 * (1 - i / size)))
    tile = Image.composite(
        ImageEnhance.Brightness(tile).enhance(1.08),
        ImageEnhance.Brightness(tile).enhance(0.94),
        shading,
    )
    return tile


def _board_background(square_size, margin):
    """خلفية الرقعة (إطار خشبي + مربعات) مع تخزين مؤقت لتسريع الرسم."""
    key = (square_size, margin)
    if key in _BOARD_BG_CACHE:
        return _BOARD_BG_CACHE[key].copy()

    board_px = square_size * 8
    total = board_px + margin * 2
    canvas = Image.new("RGB", (total, total), WOOD_FRAME)

    frame = _wood_tile(total, WOOD_FRAME, seed=999)
    canvas.paste(frame, (0, 0))

    draw = ImageDraw.Draw(canvas)
    draw.rectangle([0, 0, total - 1, total - 1], outline=WOOD_FRAME_EDGE, width=3)

    for rank in range(8):
        for file in range(8):
            base = WOOD_LIGHT if (rank + file) % 2 == 0 else WOOD_DARK
            tile = _wood_tile(square_size, base, seed=rank * 8 + file)
            if (rank + file) % 3 == 0:
                tile = tile.transpose(Image.ROTATE_180)
            canvas.paste(tile, (margin + file * square_size, margin + rank * square_size))

    # ظل داخلي حول الرقعة ليبدو الإطار بارزاً
    draw.rectangle(
        [margin - 3, margin - 3, margin + board_px + 2, margin + board_px + 2],
        outline=(52, 32, 18), width=3,
    )
    draw.rectangle(
        [margin - 1, margin - 1, margin + board_px, margin + board_px],
        outline=(158, 116, 74), width=1,
    )

    _BOARD_BG_CACHE[key] = canvas
    return canvas.copy()


def render_puzzle_board(board, flipped=False, last_move=None):
    try:
        square_size = 96
        margin = 34
        board_px = square_size * 8
        canvas = _board_background(square_size, margin)
        overlay = Image.new("RGBA", canvas.size, (0, 0, 0, 0))
        overlay_draw = ImageDraw.Draw(overlay)

        def to_xy(square):
            file_index = chess.square_file(square)
            rank_index = chess.square_rank(square)
            if flipped:
                file_index = 7 - file_index
                rank_index = 7 - rank_index
            return margin + file_index * square_size, margin + (7 - rank_index) * square_size

        # إبراز آخر نقلة
        if last_move is not None:
            for square in (last_move.from_square, last_move.to_square):
                x, y = to_xy(square)
                overlay_draw.rectangle(
                    [x, y, x + square_size - 1, y + square_size - 1],
                    fill=HIGHLIGHT + (110,),
                )

        # إبراز الملك في حالة كش
        if board.is_check():
            king_square = board.king(board.turn)
            if king_square is not None:
                x, y = to_xy(king_square)
                overlay_draw.ellipse(
                    [x + 4, y + 4, x + square_size - 5, y + square_size - 5],
                    fill=(220, 60, 50, 90),
                )

        canvas = Image.alpha_composite(canvas.convert("RGBA"), overlay)

        piece_px = int(square_size * 0.86)
        offset = (square_size - piece_px) // 2
        font_fallback = None

        for square, piece in board.piece_map().items():
            x, y = to_xy(square)
            sprite = _piece_sprite(piece.symbol(), piece_px)
            if sprite is not None:
                canvas.alpha_composite(sprite, (x + offset - 6, y + offset - 4))
            else:
                if font_fallback is None:
                    font_fallback = _load_piece_font(int(square_size * 0.82))
                font, unicode_ok = font_fallback
                symbol = UNICODE_PIECES[piece.symbol().upper()] if unicode_ok else piece.symbol().upper()
                fill = (250, 250, 250) if piece.color == chess.WHITE else (25, 25, 25)
                stroke = (25, 25, 25) if piece.color == chess.WHITE else (240, 240, 240)
                draw = ImageDraw.Draw(canvas)
                bbox = draw.textbbox((0, 0), symbol, font=font)
                tx = x + (square_size - (bbox[2] - bbox[0])) / 2 - bbox[0]
                ty = y + (square_size - (bbox[3] - bbox[1])) / 2 - bbox[1]
                draw.text((tx, ty), symbol, font=font, fill=fill, stroke_width=2, stroke_fill=stroke)

        # إحداثيات محفورة على الإطار
        label_font, _ = _load_piece_font(int(margin * 0.62))
        label_draw = ImageDraw.Draw(canvas)
        for i in range(8):
            letter = chess.FILE_NAMES[7 - i if flipped else i]
            number = str((i + 1) if flipped else (8 - i))
            cx = margin + i * square_size + square_size / 2
            cy = margin + i * square_size + square_size / 2
            for text, pos in (
                (letter, (cx, margin + board_px + margin / 2)),
                (letter, (cx, margin / 2)),
                (number, (margin / 2, cy)),
                (number, (margin + board_px + margin / 2, cy)),
            ):
                label_draw.text(
                    pos, text, font=label_font, fill=(238, 216, 186),
                    anchor="mm", stroke_width=1, stroke_fill=(48, 30, 16),
                )

        output = io.BytesIO()
        canvas.convert("RGB").save(output, format="JPEG", quality=92, subsampling=0)
        output.name = "puzzle.jpg"
        output.seek(0)
        return output
    except Exception as exc:
        log.exception("Puzzle render error: %s", exc)
        return None

def format_solution(data):
    """يحوّل نقلات الحل من صيغة UCI (b5d6) إلى الصيغة المقروءة SAN (Nxd6)."""
    solution = (data or {}).get("solution") or []
    if not solution:
        return "لا يوجد"

    start_fen = (data or {}).get("start_fen")
    if not start_fen:
        return " ➡️ ".join(solution)

    try:
        board = chess.Board(start_fen)
    except ValueError:
        return " ➡️ ".join(solution)

    parts = []
    for index, uci in enumerate(solution):
        try:
            move = chess.Move.from_uci(uci)
            san = board.san(move)
            board.push(move)
        except (ValueError, AssertionError):
            parts.extend(solution[index:])
            break
        # نقلات اللاعب في المواقع الزوجية، وردود الخصم في الفردية
        parts.append(san if index % 2 == 0 else f"({san})")
    return " ➡️ ".join(parts)


def finish_puzzle(chat_id):
    if chat_id in puzzles:
        del puzzles[chat_id]
        save_puzzles()


def get_active_puzzle(chat_id):
    """يُرجع اللغز النشط أو None، مع تنظيف الألغاز المنتهية زمنياً."""
    data = puzzles.get(chat_id)
    if not isinstance(data, dict) or data.get("status") != "active":
        return None
    elapsed = (time.time() - float(data.get("started_at", 0))) / 60
    if elapsed > float(data.get("time_limit", 1)):
        finish_puzzle(chat_id)
        return None
    return data

_ELO_CACHE = {}
_ELO_TTL = 300
USERNAME_RE = re.compile(r"^[A-Za-z0-9_-]{3,25}$")

NA_RATING = {"rapid": "N/A", "blitz": "N/A", "bullet": "N/A"}


def fetch_elo(username):
    now = time.time()
    cached = _ELO_CACHE.get(username)
    if cached and now - cached[0] < _ELO_TTL:
        return cached[1]

    url = f"https://api.chess.com/pub/player/{username}/stats"
    headers = {"User-Agent": USER_AGENT}
    try:
        response = requests.get(url, headers=headers, timeout=HTTP_TIMEOUT)
        if response.status_code != 200:
            return dict(NA_RATING)
        data = response.json()
    except (requests.RequestException, ValueError) as exc:
        log.warning("chess.com stats error for %s: %s", username, exc)
        return {"rapid": "Error", "blitz": "Error", "bullet": "Error"}

    def safe(key):
        value = data.get(key)
        if isinstance(value, dict):
            last = value.get("last")
            if isinstance(last, dict):
                return last.get("rating", "N/A")
        return "N/A"

    result = {
        "rapid": safe("chess_rapid"),
        "blitz": safe("chess_blitz"),
        "bullet": safe("chess_bullet"),
    }
    _ELO_CACHE[username] = (now, result)
    return result


async def get_elo_async(username):
    return await asyncio.to_thread(fetch_elo, username)


def fetch_games(username):
    url = f"https://api.chess.com/pub/player/{username}/games/archives"
    headers = {"User-Agent": USER_AGENT}
    try:
        response = requests.get(url, headers=headers, timeout=HTTP_TIMEOUT)
        if response.status_code != 200:
            return []
        archives = response.json().get("archives", [])
    except (requests.RequestException, ValueError):
        return []

    all_games = []
    for archive in archives[-2:]:
        try:
            game_request = requests.get(archive, headers=headers, timeout=HTTP_TIMEOUT)
            if game_request.status_code == 200:
                all_games.extend(game_request.json().get("games", []))
        except (requests.RequestException, ValueError):
            continue
    return all_games


def fetch_match_result(player1, player2, challenge_time):
    games = fetch_games(player1)
    for game in reversed(games):
        try:
            if int(game.get("end_time", 0)) < int(challenge_time):
                continue
            white = game["white"]["username"].lower()
            black = game["black"]["username"].lower()
        except (KeyError, TypeError, ValueError):
            continue
        if {white, black} != {player1.lower(), player2.lower()}:
            continue
        if game["white"].get("result") == "win":
            return {"winner": white, "game_url": game.get("url", "")}
        if game["black"].get("result") == "win":
            return {"winner": black, "game_url": game.get("url", "")}
        return {"winner": "draw", "game_url": game.get("url", "")}
    return None


async def get_match_result_async(player1, player2, challenge_time):
    return await asyncio.to_thread(fetch_match_result, player1, player2, challenge_time)

def schedule_puzzle_timeout(context, chat_id, minutes):
    job_queue = getattr(context, "job_queue", None)
    if job_queue is None:
        return
    name = f"puzzle_timeout_{chat_id}"
    for job in job_queue.get_jobs_by_name(name):
        job.schedule_removal()
    job_queue.run_once(puzzle_timeout_job, minutes * 60, chat_id=chat_id, name=name)

def parse_user_move(board, raw_text):
    """يقبل SAN (Rc5) أو UCI (c1c5)، مع أو بدون / في البداية."""
    text = raw_text.strip()
    if text.startswith("/"):
        text = text[1:]
    text = text.split("@")[0].strip()
    if not text or len(text) > 8:
        return None
    for parser in (board.parse_san, board.parse_uci):
        try:
            return parser(text)
        except (ValueError, chess.InvalidMoveError,
                chess.IllegalMoveError, chess.AmbiguousMoveError):
            continue
    return None


async def handle_puzzle_reply(update: Update, context: ContextTypes.DEFAULT_TYPE):
    set_active_chat(update.effective_chat.id)
    chat_id = str(update.effective_chat.id)
    msg = update.effective_message
    if msg is None or not msg.text:
        return

    current = puzzles.get(chat_id)
    if not isinstance(current, dict) or current.get("status") != "active":
        return

    if msg.from_user is None or msg.from_user.id != current.get("owner_id"):
        return

    elapsed = (time.time() - float(current.get("started_at", 0))) / 60
    if elapsed > float(current.get("time_limit", 1)):
        solution_text = format_solution(current)
        finish_puzzle(chat_id)
        await msg.reply_text(
            "⏱️ <b>انتهى الوقت المحدد للغز!</b>\n"
            f"📌 الحل كان:\n<code>{escape(solution_text)}</code>",
            parse_mode="HTML",
        )
        return

    board = chess.Board(current["fen"])
    user_move = parse_user_move(board, msg.text)
    if user_move is None:
        return  # ليست نقلة — تجاهل بصمت

    solution = current.get("solution") or []
    step = int(current.get("current_step", 0))
    if step >= len(solution):
        finish_puzzle(chat_id)
        return

    try:
        expected_move = chess.Move.from_uci(solution[step])
    except ValueError:
        finish_puzzle(chat_id)
        await msg.reply_text("❌ بيانات اللغز تالفة، تم إلغاؤه. جرّب /puzzle مجدداً.")
        return

    player_white = bool(current.get("player_white", True))

    if user_move != expected_move:
        current["wrong_attempts"] = int(current.get("wrong_attempts", 0)) + 1
        if current["wrong_attempts"] >= MAX_WRONG_ATTEMPTS:
            solution_text = format_solution(current)
            finish_puzzle(chat_id)
            await msg.reply_text(
                f"❌ <b>إجابة خاطئة!</b> استنفدت محاولاتك ({MAX_WRONG_ATTEMPTS}/{MAX_WRONG_ATTEMPTS}).\n"
                f"🏁 انتهى اللغز. الحل الصحيح:\n<code>{escape(solution_text)}</code>",
                parse_mode="HTML",
            )
        else:
            puzzles[chat_id] = current
            save_puzzles()
            remaining = MAX_WRONG_ATTEMPTS - current["wrong_attempts"]
            await msg.reply_text(
                f"⚠️ <b>إجابة خاطئة!</b> تبقّى لديك {remaining} محاولة.",
                parse_mode="HTML",
            )
        return

    # نقلة صحيحة
    board.push(user_move)
    step += 1
    last_move = user_move
    reply_san = None

    if step < len(solution):
        try:
            reply_move = chess.Move.from_uci(solution[step])
            reply_san = board.san(reply_move)
            board.push(reply_move)
            last_move = reply_move
            step += 1
        except (ValueError, AssertionError):
            step = len(solution)

    current["current_step"] = step
    current["fen"] = board.fen()

    if step >= len(solution):
        finish_puzzle(chat_id)
        caption = "🎉 <b>أحسنت! تم حل اللغز بالكامل بنجاح.</b> 🏆"
    else:
        puzzles[chat_id] = current
        save_puzzles()
        caption = (
            f"✅ <b>نقلة صحيحة!</b>\n"
            f"🤖 رد الخصم: <code>{escape(reply_san or '')}</code>\n"
            "ما هي نقلتك التالية؟"
        )

    board_img = render_puzzle_board(board, flipped=not player_white, last_move=last_move)
    if board_img is None:
        await msg.reply_text(caption, parse_mode="HTML")
    else:
        await msg.reply_photo(photo=board_img, caption=caption, parse_mode="HTML")

POINTS_RE = re.compile(r"^([+\-−])\s*(\d{1,4})$")


async def handle_points(update: Update, context: ContextTypes.DEFAULT_TYPE):
    set_active_chat(update.effective_chat.id)
    msg = update.effective_message
    if msg is None or not msg.text or not msg.reply_to_message:
        return

    match = POINTS_RE.match(msg.text.strip())
    if not match:
        return
    sign_char, amount = match.group(1), int(match.group(2))
    if amount == 0:
        return
    value = -amount if sign_char in "-−" else amount

    admin_id = update.effective_user.id
    if not await is_admin(update.effective_chat, admin_id) and not is_point_admin(admin_id):
        return

    target = msg.reply_to_message.from_user
    if target is None or target.is_bot:
        return
    uid = str(target.id)
    new_score = max(0, get_score(uid) + value)
    points[uid] = new_score
    save_points()
    await msg.reply_text(
        f"🏆 <b>تم تعديل النقاط</b>\n\n👤 الاسم: {escape(get_name(target))}\n"
        f"📊 النقاط: {new_score}\n🎖️ الرتبة: {escape(get_rank(new_score))}",
        parse_mode="HTML",
    )

def sorted_points(limit=10):
    rows = []
    for uid, value in points.items():
        try:
            rows.append((uid, int(value)))
        except (TypeError, ValueError):
            continue

    return rows[:limit]

async def result(update: Update, context: ContextTypes.DEFAULT_TYPE):
    set_active_chat(update.effective_chat.id)
    msg = update.effective_message
    if not msg.reply_to_message:
        await msg.reply_text("❌ استخدم الأمر بالرد على رسالة التحدي.")
        return
    if (not await is_admin(update.effective_chat, update.effective_user.id)
            and not is_point_admin(update.effective_user.id)):
        await msg.reply_text("❌ هذا الأمر للمشرفين ومسؤولي النقاط فقط.")
        return

    reply_id = msg.reply_to_message.message_id
    match_id, match = None, None
    for key, data in matches.items():
        if isinstance(data, dict) and data.get("message_id") == reply_id:
            match_id, match = key, data
            break
    if match is None:
        await msg.reply_text("❌ هذه ليست رسالة تحدي.")
        return
    if match.get("status") == "finished":
        await msg.reply_text("❌ تم اعتماد نتيجة هذه المباراة مسبقًا.")
        return

    notice = await msg.reply_text("⏳ جاري التحقق من المباراة على chess.com...")
    res = await get_match_result_async(match["username1"], match["username2"], match["time"])
    if res is None:
        await notice.edit_text("⏳ لم يتم العثور على مباراة منتهية بين اللاعبين بعد.")
        return

    if res["winner"] == "draw":
        match["status"], match["result"] = "finished", "draw"
        matches[match_id] = match
        save_matches()
        await edit_match_message(
            context, update.effective_chat.id, match["message_id"],
            f"🤝 <b>انتهت المباراة</b>\n━━━━━━━━━━━━━━\n\n⚖️ <b>النتيجة:</b> تعادل\n\n"
            f"🔗 <a href=\"{escape(res['game_url'])}\">عرض المباراة</a>\n\n✅ تم اعتماد النتيجة رسميًا.",
        )
        await notice.edit_text("✅ تم اعتماد النتيجة (تعادل).")
        return

    if res["winner"] == str(match["username1"]).lower():
        winner_id, loser_id = match["player1"], match["player2"]
        winner_name, loser_name = match["username1"], match["username2"]
    else:
        winner_id, loser_id = match["player2"], match["player1"]
        winner_name, loser_name = match["username2"], match["username1"]

    points[winner_id] = get_score(winner_id) + WIN_POINTS
    loser_old = get_score(loser_id)
    lose_points = 8 if loser_old >= 200 else (5 if loser_old >= 100 else 0)
    points[loser_id] = max(0, loser_old - lose_points)
    save_points()

    today = datetime.now().strftime("%Y-%m-%d")
    day = daily_wins.get(today)
    if not isinstance(day, dict):
        day = {}
    day[winner_id] = int(day.get(winner_id, 0)) + 1
    daily_wins[today] = day
    save_daily_wins()

    match["status"], match["result"] = "finished", res["winner"]
    matches[match_id] = match
    save_matches()

    await edit_match_message(
        context, update.effective_chat.id, match["message_id"],
        f"🏁 <b>انتهت المباراة</b>\n━━━━━━━━━━━━━━\n\n👑 <b>الفائز</b>\n<code>{escape(str(winner_name))}</code>\n"
        f"🏆 <b>المكافأة:</b> +{WIN_POINTS} نقطة\n\n📉 <b>الخاسر</b>\n<code>{escape(str(loser_name))}</code>\n"
        f"➖ <b>الخصم:</b> {lose_points} نقطة\n\n"
        f"🔗 <a href=\"{escape(res['game_url'])}\">عرض المباراة</a>\n\n✅ تم اعتماد النتيجة رسميًا.",
    )
    await notice.edit_text("✅ تم اعتماد النتيجة وتحديث النقاط بنجاح.")

        await safe_edit(query, "❌ لا يوجد لاعبين مربوطين.", reply_markup=back_keyboard())

        *(get_elo_async(username) for _, username in linked), return_exceptions=True

        name = await resolve_display_name(context.bot, query.message.chat.id, uid)
        players.append((value, name, username))

    if not players:
        await safe_edit(query, "❌ لا توجد تصنيفات متاحة حالياً.", reply_markup=back_keyboard())
        return

    players.sort(key=lambda item: item[0], reverse=True)
    text = f"{icon} <b>{title}</b>\n━━━━━━━━━━━━━━━━━━\n\n"
    for index, (value, name, chess_un) in enumerate(players[:10]):
        place = PLACES[index]
        text += (
            f"{place} <b>{escape(name)}</b>\n"
            f"\u265f\ufe0f <code>{escape(str(chess_un))}</code>\n"
            f"{icon} {key.title()} : <b>{value}</b>\n\n"
        )
    text += (
        f"━━━━━━━━━━━━━━━━━━\n👑 <b>{king_label}:</b> "
        f"{escape(players[0][1])} • <b>{players[0][0]}</b>"
    )
    await safe_edit(query, text, parse_mode="HTML", reply_markup=back_keyboard())

    set_active_chat(update.effective_chat.id)

    if update.effective_user.id != OWNER_ID:
        await msg.reply_text("❌ هذا الأمر متاح للمالك فقط.")

# شكل نقلة SAN أو UCI فقط (يمنع التقاط رسائل مثل "+10" أو كلمات عادية)
MOVE_SHAPE_RE = re.compile(
    r"(?:[Oo0](?:-[Oo0]){1,2}|[KQRBN]?[a-h]?[1-8]?x?[a-h][1-8](?:=[QRBN])?[+#]?)"
)


class PuzzleMoveFilter(filters.MessageFilter):
    """يمرّر فقط الرسائل التي قد تكون نقلة شطرنج (وليست أمراً معروفاً)."""

    def filter(self, message):
        text = (message.text or "").strip()
        if not text or len(text) > 10:
            return False
        candidate = text[1:] if text.startswith("/") else text
        candidate = candidate.split("@")[0].strip()
        if not candidate or candidate.lower() in KNOWN_COMMANDS:
            return False
        return bool(MOVE_SHAPE_RE.fullmatch(candidate))
