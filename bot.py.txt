def get_elo(username):
    url = f"https://api.chess.com/pub/player/{username}/stats"
    headers = {"User-Agent": "TelegramChessBot/1.0 (contact: admin@example.com)"}

    try:
        r = requests.get(url, headers=headers, timeout=10)
        if r.status_code != 200:
            return {"rapid": "N/A", "blitz": "N/A", "bullet": "N/A"}
        data = r.json()
    except Exception:
        return {"rapid": "Error", "blitz": "Error", "bullet": "Error"}

    def safe(key):
        try:
            return data.get(key, {}).get("last", {}).get("rating", "N/A")
        except Exception:
            return "N/A"

    return {
        "rapid": safe("chess_rapid"),
        "blitz": safe("chess_blitz"),
        "bullet": safe("chess_bullet"),
    }

def get_games(username):
    url = f"https://api.chess.com/pub/player/{username}/games/archives"
    try:
        r = requests.get(url, timeout=10)
        if r.status_code != 200:
            return []
        archives = r.json().get("archives", [])
        all_games = []
        for archive in archives:
            try:
                game_request = requests.get(archive, timeout=10)
                if game_request.status_code == 200:
                    games = game_request.json().get("games", [])
                    all_games.extend(games)
            except Exception:
                continue
        return all_games
    except Exception:
        return []

def get_match_result(player1, player2, challenge_time):
    games = get_games(player1)
    for game in reversed(games):
        end_time = game.get("end_time", 0)
        if end_time < challenge_time:
            continue

        white = game["white"]["username"].lower()
        black = game["black"]["username"].lower()

        if {white, black} != {player1.lower(), player2.lower()}:
            continue

        if game["white"]["result"] == "win":
            return {"winner": white, "game_url": game.get("url", "")}

        if game["black"]["result"] == "win":
            return {"winner": black, "game_url": game.get("url", "")}

        return {"winner": "draw", "game_url": game.get("url", "")}

    return None
