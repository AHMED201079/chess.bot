_GROUP_KEY = "__groups__"

def set_active_chat(chat_id):

class GroupDict:
    """Dictionary facade: every group gets its own independent namespace."""
    def __init__(self, filename, default_factory=dict):
        self.filename = filename
        self.default_factory = default_factory
        self.data = self._load()

    def _load(self):
        if not os.path.exists(self.filename):
            return {_GROUP_KEY: {}}
        try:
            with open(self.filename, "r", encoding="utf-8") as f:
                raw = json.load(f)
        except Exception:
            return {_GROUP_KEY: {}}

        # Old version was a flat dictionary. Keep it isolated as legacy data
        # instead of accidentally sharing it with every group.
        if not isinstance(raw, dict) or _GROUP_KEY not in raw or not isinstance(raw.get(_GROUP_KEY), dict):
            return {_GROUP_KEY: {}, "__legacy__": raw}
        return raw

    def _group(self):
        gid = get_active_chat_id()
        groups = self.data.setdefault(_GROUP_KEY, {})
        if gid not in groups:
            groups[gid] = self.default_factory()
        return groups[gid]

    def save(self):
        with open(self.filename, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=4)

    def get(self, key, default=None):
        return self._group().get(key, default)

    def items(self):
        return self._group().items()

    def values(self):
        return self._group().values()

    def keys(self):
        return self._group().keys()

    def __contains__(self, key):
        return key in self._group()

    def __getitem__(self, key):
        return self._group()[key]

    def __setitem__(self, key, value):
        self._group()[key] = value

    def __delitem__(self, key):
        del self._group()[key]

    def __bool__(self):
        return bool(self._group())

    def __len__(self):
        return len(self._group())

class GroupList(GroupDict):
    def __init__(self, filename):
        super().__init__(filename, list)

    def _group(self):
        gid = get_active_chat_id()
        groups = self.data.setdefault(_GROUP_KEY, {})
        if gid not in groups:
            groups[gid] = []
        return groups[gid]

    def append(self, value):
        self._group().append(value)

    def remove(self, value):
        self._group().remove(value)

    def __iter__(self):
        return iter(self._group())

def migrate_legacy_here():
    """One-time owner command: import the old flat data into THIS group only."""
    gid = get_active_chat_id()
    changed = False

    for store in (users, points, matches, daily_wins, badges):
        legacy = store.data.get("__legacy__")
        if legacy and isinstance(legacy, dict):
            store.data.setdefault(_GROUP_KEY, {})[gid] = legacy
            del store.data["__legacy__"]
            store.save()
            changed = True

    legacy = point_admins.data.get("__legacy__")
    if isinstance(legacy, list):
        point_admins.data.setdefault(_GROUP_KEY, {})[gid] = legacy
        del point_admins.data["__legacy__"]
        point_admins.save()
        changed = True

    return changed


def get_elo(username):
    url = f"https://api.chess.com/pub/player/{username}/stats"
    headers = {
        "User-Agent": "TelegramChessBot/1.0 (contact: admin@example.com)"
    }

    try:
        r = requests.get(url, headers=headers, timeout=10)
        if r.status_code != 200:
            return {"rapid": "N/A", "blitz": "N/A", "bullet": "N/A"}
        data = r.json()
    except:
        return {"rapid": "Error", "blitz": "Error", "bullet": "Error"}

    def safe(key):
        try:
            return data.get(key, {}).get("last", {}).get("rating", "N/A")
        except:
            return "N/A"

    return {
        "rapid": safe("chess_rapid"),
        "blitz": safe("chess_blitz"),
        "bullet": safe("chess_bullet")
    }

def get_games(username):
    url = f"https://api.chess.com/pub/player/{username}/games/archives"

    try:
        r = requests.get(url, timeout=10)

        if r.status_code != 200:
            return []

        archives = r.json().get("archives", [])

        all_games = []

        for archive in archives:
            try:
                game_request = requests.get(archive, timeout=10)

                if game_request.status_code == 200:
                    games = game_request.json().get("games", [])
                    all_games.extend(games)

            except:
                continue

        return all_games

    except:
        return []

def get_chess_stats(username):

    games = get_games(username)

    wins = 0
    losses = 0
    draws = 0
    total = 0

    for game in games:

        try:
            white = game["white"]["username"].lower()
            black = game["black"]["username"].lower()

            if username.lower() == white:
                result = game["white"]["result"]

            elif username.lower() == black:
                result = game["black"]["result"]

            else:
                continue

            total += 1

            if result == "win":
                wins += 1

            elif result in [
                "checkmated",
                "resigned",
                "timeout",
                "lose"
            ]:
                losses += 1

            elif result == "draw":
                draws += 1

        except:
            continue


    if total:
        rate = round((wins / total) * 100)
    else:
        rate = 0


    return {
        "games": total,
        "wins": wins,
        "draws": draws,
        "losses": losses,
        "rate": rate
    }


def get_match_result(player1, player2, challenge_time):

    games = get_games(player1)

    for game in reversed(games):

        end_time = game.get("end_time", 0)

        if end_time < challenge_time:
            continue

        white = game["white"]["username"].lower()
        black = game["black"]["username"].lower()

        if {white, black} != {player1.lower(), player2.lower()}:
            continue

        if game["white"]["result"] == "win":
            return {
                "winner": white,
                "game_url": game.get("url", "")
            }

        if game["black"]["result"] == "win":
            return {
                "winner": black,
                "game_url": game.get("url", "")
            }

        return {
            "winner": "draw",
            "game_url": game.get("url", "")
        }

    return None

    if challenger_id == opponent_id:
        await update.message.reply_text("❌ لا يمكنك تحدي نفسك!")
        return

    # التحقق من أن المتحدي رابط حسابه
    if challenger_id not in users:
        await update.message.reply_text("❌ يجب ربط حسابك أولاً باستخدام /sign")
        return

    # التحقق من أن الطرف الآخر رابط حسابه
    if opponent_id not in users:
        await update.message.reply_text("❌ هذا اللاعب غير مربوط بحساب Chess.com.")
        return

    import time

    match_id = str(int(time.time()))

    matches[match_id] = {
    "player1": challenger_id,
    "player2": opponent_id,
    "username1": users[challenger_id],
    "username2": users[opponent_id],
    "message_id": 0,
    "time": int(time.time()),
    "status": "active",
    "results": []
}

    save_matches()

    challenger_chess = users[challenger_id]
    opponent_chess = users[opponent_id]

    challenge_url = f"https://www.chess.com/play/online/new?opponent={opponent_chess}"

    text = (
    f"⚔️ <b>تحدي شطرنج جديد</b>\n"
    f"━━━━━━━━━━━━━━\n\n"
    f"👤 <b>{escape(get_name(challenger))}</b>\n"
    f"♟️ <code>{challenger_chess}</code>\n\n"
    f"🆚\n\n"
    f"👤 <b>{escape(get_name(opponent))}</b>\n"
    f"♟️ <code>{opponent_chess}</code>\n\n"
    f"━━━━━━━━━━━━━━\n"
    f"🔗 <a href='{challenge_url}'><b>ابدأ التحدي</b></a>\n\n"
    f"🏆 بعد انتهاء المباراة استخدم <code>/result</code> لاعتماد النتيجة."
)


    msg = await update.message.reply_text(
    text,
    parse_mode="HTML",
    disable_web_page_preview=False
   )

    matches[match_id]["message_id"] =          msg.message_id
    save_matches()

async def result(update: Update, context: ContextTypes.DEFAULT_TYPE):
    set_active_chat(update.effective_chat.id)

    if not update.message.reply_to_message:
        await update.message.reply_text("❌ استخدم الأمر بالرد على رسالة التحدي.")
        return

    if not await is_admin(update, update.effective_user.id) and str(update.effective_user.id) not in point_admins:
        await update.message.reply_text("❌ هذا الأمر للمشرفين ومسؤولي النقاط فقط.")
        return

    reply_id = update.message.reply_to_message.message_id

    match = None

    for data in matches.values():
        if data.get("message_id") == reply_id:
            match = data
            break

    if not match:
        await update.message.reply_text("❌ هذه ليست رسالة تحدي.")
        return

    result = get_match_result(
        match["username1"],
        match["username2"],
        match["time"]
    )

    print(match["username1"])
    print(match["username2"])
    print(match["time"])
    print(result)

    if result is None:
        await update.message.reply_text(
            "⏳ لم يتم العثور على مباراة بين اللاعبين بعد."
        )
        return 

    if match["status"] == "finished":
        await update.message.reply_text(
            "❌ تم اعتماد نتيجة هذه المباراة مسبقًا."
        )
        return

    # التعادل
    if result["winner"] == "draw":
        match["status"] = "finished"
        match["result"] = "draw"
        save_matches()

        await context.bot.edit_message_text(
            chat_id=update.effective_chat.id,
            message_id=match["message_id"],
            parse_mode="HTML",
            disable_web_page_preview=False,
            text=f"""
🤝 <b>انتهت المباراة</b>
━━━━━━━━━━━━━━

⚖️ <b>النتيجة:</b> تعادل

🔗 <a href="{result['game_url']}">عرض المباراة</a>

✅ تم اعتماد النتيجة رسميًا.
"""
        )

        await update.message.reply_text("✅ تم اعتماد نتيجة التعادل.")
        return

    # تحديد الفائز والخاسر
    if result["winner"] == match["username1"].lower():
        winner_id = match["player1"]
        loser_id = match["player2"]
        winner_name = match["username1"]
        loser_name = match["username2"]
    else:
        winner_id = match["player2"]
        loser_id = match["player1"]
        winner_name = match["username2"]
        loser_name = match["username1"]

    # نقاط الفائز
    winner_old = points.get(winner_id, 20)
    points[winner_id] = winner_old + 8

    # خصم نقاط الخاسر
    loser_old = points.get(loser_id, 20)

    if loser_old >= 200:
        lose_points = 8
    elif loser_old >= 100:
        lose_points = 5
    else:
        lose_points = 0

    points[loser_id] = max(0, loser_old - lose_points)

    save_points()

    # عدد الانتصارات اليومية
    today = datetime.now().strftime("%Y-%m-%d")

    if today not in daily_wins:
        daily_wins[today] = {}

    daily_wins[today][winner_id] = daily_wins[today].get(winner_id, 0) + 1

    save_daily_wins()

    match["status"] = "finished"
    match["result"] = result["winner"]

    save_matches()

    await context.bot.edit_message_text(
        chat_id=update.effective_chat.id,
        message_id=match["message_id"],
        parse_mode="HTML",
        disable_web_page_preview=False,
        text=f"""
🏁 <b>انتهت المباراة</b>
━━━━━━━━━━━━━━

👑 <b>الفائز</b>
<code>{winner_name}</code>

🏆 <b>المكافأة:</b> +8 نقطة

📉 <b>الخاسر</b>
<code>{loser_name}</code>

➖ <b>الخصم:</b> {lose_points} نقطة

🔗 <a href="{result['game_url']}">عرض المباراة</a>

✅ تم اعتماد النتيجة رسميًا.
"""
    )

    await update.message.reply_text(
        "✅ تم اعتماد النتيجة وتحديث النقاط بنجاح."
    )

        badges[uid][index] = existing + " ★"

    else:
        badges[uid].append(badge)


    save_badges()


    await update.message.reply_text(
f"""

{get_name(user)}

{badge}

)
