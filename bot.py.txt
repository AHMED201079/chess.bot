def get_elo(username):
    url = f"https://api.chess.com/pub/player/{username}/stats"
    headers = {
        "User-Agent": "TelegramChessBot/1.0 (contact: admin@example.com)"
    }

    try:
        r = requests.get(url, headers=headers, timeout=10)
        if r.status_code != 200:
            return {"rapid": "N/A", "blitz": "N/A", "bullet": "N/A"}
        data = r.json()
    except:
        return {"rapid": "Error", "blitz": "Error", "bullet": "Error"}

    def safe(key):
        try:
            return data.get(key, {}).get("last", {}).get("rating", "N/A")
        except:
            return "N/A"

    return {
        "rapid": safe("chess_rapid"),
        "blitz": safe("chess_blitz"),
        "bullet": safe("chess_bullet")
    }

def get_games(username):
    url = f"https://api.chess.com/pub/player/{username}/games/archives"

    try:
        r = requests.get(url, timeout=10)
        if r.status_code != 200:
            return []

        archives = r.json().get("archives", [])

        all_games = []

        for archive in archives:
            try:
                r = requests.get(archive, timeout=10)
                if r.status_code == 200:
                    all_games.extend(r.json().get("games", []))
            except:
                pass

        return all_games

    except:
        return []

   def get_match_result(player1, player2, challenge_time):

    games = get_games(player1)

    for game in reversed(games):

        end_time = game.get("end_time", 0)

        if end_time < challenge_time:
            continue

        white = game["white"]["username"].lower()
        black = game["black"]["username"].lower()

        if {white, black} != {player1.lower(), player2.lower()}:
            continue

        if game["white"]["result"] == "win":
            return {
                "winner": white,
                "game_url": game.get("url", "")
            }

        if game["black"]["result"] == "win":
            return {
                "winner": black,
                "game_url": game.get("url", "")
            }

        return {
            "winner": "draw",
            "game_url": game.get("url", "")
        }

    return None

async def play(update: Update, context: ContextTypes.DEFAULT_TYPE):
    # التأكد من أن الأمر أُرسل بالرد على رسالة
    if not update.message.reply_to_message:
        await update.message.reply_text("❌ استخدم الأمر بالرد على رسالة اللاعب الذي تريد تحديه.")
        return

    challenger = update.effective_user
    opponent = update.message.reply_to_message.from_user

    challenger_id = str(challenger.id)
    opponent_id = str(opponent.id)

    # منع التحدي مع البوتات أو منع تحدي الشخص لنفسه
    if challenger.is_bot or opponent.is_bot:
        return

    if challenger_id == opponent_id:
        await update.message.reply_text("❌ لا يمكنك تحدي نفسك!")
        return

    # التحقق من أن المتحدي رابط حسابه
    if challenger_id not in users:
        await update.message.reply_text("❌ يجب ربط حسابك أولاً باستخدام /sign")
        return

        await update.message.reply_text("❌ هذا اللاعب غير مربوط بحساب Chess.com.")
        return

    import time

    match_id = str(int(time.time()))

    matches[match_id] = {
    "player1": challenger_id,
    "player2": opponent_id,
    "username1": users[challenger_id],
    "username2": users[opponent_id],
    "message_id": 0,
    "time": int(time.time()),
    "status": "active",
    "results": []
}

    save_matches()

    challenger_chess = users[challenger_id]
    opponent_chess = users[opponent_id]

    challenge_url = f"https://www.chess.com/play/online/new?opponent={opponent_chess}"

    text = (
    f"⚔️ <b>تحدي شطرنج جديد</b>\n"
    f"━━━━━━━━━━━━━━\n\n"
    f"👤 <b>{escape(get_name(challenger))}</b>\n"
    f"♟️ <code>{challenger_chess}</code>\n\n"
    f"🆚\n\n"
    f"👤 <b>{escape(get_name(opponent))}</b>\n"
    f"♟️ <code>{opponent_chess}</code>\n\n"
    f"━━━━━━━━━━━━━━\n"
    f"🔗 <a href='{challenge_url}'><b>ابدأ التحدي</b></a>\n\n"
    f"🏆 بعد انتهاء المباراة استخدم <code>/result</code> لاعتماد النتيجة."
)


    msg = await update.message.reply_text(
    text,
    parse_mode="HTML",
    disable_web_page_preview=False
   )

    matches[match_id]["message_id"] =          msg.message_id
    save_matches()

        return

    result = get_match_result(
        match["username1"],
        match["username2"],
        match["time"]
    )

    print(match["username2"])
    print(match["time"])
    print(result)

    if result is None:
        await update.message.reply_text(
            "⏳ لم يتم العثور على مباراة بين اللاعبين بعد."

    # تحديد الفائز والخاسر
    if result["winner"] == match["username1"].lower():
        winner_id = match["player1"]
        loser_id = match["player2"]
        winner_name = match["username1"]
        loser_name = match["username2"]
    else:
        winner_id = match["player2"]
        loser_id = match["player1"]
        winner_name = match["username2"]
        loser_name = match["username1"]

    # نقاط الفائز
    winner_old = points.get(winner_id, 20)

        lose_points = 8
    elif loser_old >= 100:
        lose_points = 5
    else:
        lose_points = 0

    points[loser_id] = max(0, loser_old - lose_points)

    save_points()

    # عدد الانتصارات اليومية
    today = datetime.now().strftime("%Y-%m-%d")

    if today not in daily_wins:
        daily_wins[today] = {}

    match["result"] = result["winner"]

    await context.bot.edit_message_text(
        chat_id=update.effective_chat.id,
        message_id=match["message_id"],
        parse_mode="HTML",
        disable_web_page_preview=False,
        text=f"""
🏁 <b>انتهت المباراة</b>
━━━━━━━━━━━━━━

👑 <b>الفائز</b>
<code>{winner_name}</code>

🏆 <b>المكافأة:</b> +8 نقطة

📉 <b>الخاسر</b>
<code>{loser_name}</code>

➖ <b>الخصم:</b> {lose_points} نقطة

🔗 <a href="{result['game_url']}">عرض المباراة</a>

✅ تم اعتماد النتيجة رسميًا.
